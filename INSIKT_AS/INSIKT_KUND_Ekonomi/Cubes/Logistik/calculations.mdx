--
-- Ack
--
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV YTD]
 AS 
sum
(
    PeriodsToDate
    (
        [Transaktionsdatum].[Redovisningsår-Period].[Redovisningsår]
        ,
        StrToMember
        (
            "[Transaktionsdatum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'
        )        
    )  
    ,
    [Datum Calc].[Date Calc].[All]
)       
;  
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV YTD DTDAction]
 AS 
   null;

--
-- R12
--
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV R12]
AS 
sum
(
    PARALLELPERIOD(
        [Transaktionsdatum].[Redovisningsår-Period].[Redovisningsår], 1, 
        StrToMember
        (
            "[Transaktionsdatum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'
        ).Lead(1)
    )
    :
       StrToMember
       (
            "[Transaktionsdatum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'
       )        
    ,
    [Datum Calc].[Date Calc].[All]
)       
; 

--
-- Omvänd Ack. Helår - Ack

(
    {
        StrToMember
        (
            "[Transaktionsdatum].[Redovisningsår-Period].[Period].&["

            + ']'
        ).nextmember
        :      
        StrToMember
        (
            "[Transaktionsdatum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'

CREATE MEMBER CURRENTCUBE.[Planeringsversion Calc].[Pv Calc].[Prev Budget]
 AS 
	case 
	when [Planeringsversion].[Lista].properties("Previous Budget ID") = "0"
	then 
		null
	else
	(
		StrToMember
		(
			"[Planeringsversion].[Lista].[Planeringsversion].&["

			+ ']'
		)
		,[Planeringsversion Calc].[Pv Calc].[All]
	)
	end
;

   null;

            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'

    )

            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'

--
-- Ack
--
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV YTD]
 AS 
sum
(
    PeriodsToDate
    (
        [Datum].[Redovisningsår-Period].[Redovisningsår]
        ,
        StrToMember
        (
            "[Datum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'
        )        
    )  
    ,
    [Datum Calc].[Date Calc].[All]
)       
;  
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV YTD DTDAction]
 AS 
   null;

--
-- R12
--
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV R12]
AS 
sum
(
    PARALLELPERIOD(
        [Datum].[Redovisningsår-Period].[Redovisningsår], 1, 
        StrToMember
        (
            "[Datum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'
        ).Lead(1)
    )
    :
       StrToMember
       (
            "[Datum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'
       )        
    ,
    [Datum Calc].[Date Calc].[All]
)       
; 

--
-- Omvänd Ack. Helår - Ack
--
CREATE MEMBER CURRENTCUBE.[Datum Calc].[Date Calc].[PV Rev YTD]

sum
(
    {
        StrToMember
        (
            "[Datum].[Redovisningsår-Period].[Period].&["

        ).nextmember
        :      
        StrToMember
        (
            "[Datum].[Redovisningsår-Period].[Period].&["
            + [Planeringsversion].[Lista].properties("Period Month")
            + ']'

    ,

)
